15/// Function to format the rank into a plain comma-separated string for the `ranks` URL parameter.
16///
17/// Returns an empty string if the rank is not recognised (e.g. "none").
18fn format_rank(r: &str) -> String {
19    let ranks = [
20        "subspecies",
21        "species",
22        "genus",
23        "family",
24        "order",
25        "class",
26        "phylum",
27        "kingdom",
28        "superkingdom",
29    ];
30    let position_selected = ranks.iter().position(|e| e == &r);
31    match position_selected {
32        Some(p) => ranks[p..].join(","),
33        None => "".to_string(),
34    }
35}

37/// Percent-encode a URL query parameter value using `%20` for spaces.
38///
39/// Unlike `url::form_urlencoded` (which encodes spaces as `+`), this produces
40/// standard URL percent-encoding that GoaT's API requires.
41pub fn percent_encode_query_value(s: &str) -> String {
42    let mut encoded = String::with_capacity(s.len() * 3);
43    for byte in s.bytes() {
44        match byte {
45            // RFC 3986 unreserved characters — pass through unchanged
46            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'_' | b'.' | b'~' => {
47                encoded.push(byte as char);
48            }
49            // Everything else (including space → %20, < → %3C, ( → %28, etc.)
50            b => encoded.push_str(&format!("%{:02X}", b)),
51        }
52    }
53    encoded
54}

56/// Returns the plain comma-separated value for the `names` URL parameter, or empty string.
57fn format_names(flag: bool) -> String {
58    match flag {
59        true => "synonym,tol_id,common_name".to_string(),
60        false => "".to_string(),

64/// Format an expression put into the `-e` flag on the CLI.
65pub fn format_expression(exp: &str, index_type: IndexType) -> Result<String> {
66    // Pull any live fields from the dynamic registry (empty if not yet initialised).
67    let dynamic_fields = field_registry::get_all_fields(index_type);
68    let extra = if dynamic_fields.is_empty() {
69        None
70    } else {
71        Some(&dynamic_fields)
72    };
73    let mut new_exp = CLIexpression::new(exp);
74    let parsed_string = match index_type {
75        IndexType::Taxon => new_exp.parse(&GOAT_TAXON_VARIABLE_DATA, extra)?,
76        IndexType::Assembly => new_exp.parse(&GOAT_ASSEMBLY_VARIABLE_DATA, extra)?,
77    };
78    Ok(parsed_string)
79}

322    /// Returns a plain comma-separated field list (no `&fields=` prefix, no percent-encoding).
323    ///
324    /// Returns an empty string if no field flags are set.
325    pub fn build_fields_string(&self) -> String {
326        let data = self.to_vec_tuples();
327
328        let mut fields: Vec<String> = Vec::new();
329        for (field_present, field_vec) in data.iter() {
330            if *field_present {
331                for field in field_vec {
332                    fields.push(field.to_string());
333                    if self.taxon_toggle_direct {
334                        fields.push(format!("{}:direct", field));
335                        fields.push(format!("{}:ancestor", field));
336                        fields.push(format!("{}:descendant", field));
337                    }
338                }
339            }
340        }
341
342        fields.join(",")
343    }

345    /// Returns key-value pairs for excluding missing and ancestral values.
346    ///
347    /// Each pair is `(param_name, field_name)` — the URL builder handles encoding.
348    fn generate_exclude_flags(&self) -> Vec<(String, String)> {
349        let data = self.to_vec_tuples();
350        let mut pairs: Vec<(String, String)> = Vec::new();
351
352        let mut exclude_index: usize = 0;
353        for (field_present, field_vec) in data.iter() {
354            if *field_present {
355                for field in field_vec {
356                    pairs.push((
357                        format!("excludeAncestral[{}]", exclude_index),
358                        field.to_string(),
359                    ));
360                    pairs.push((
361                        format!("excludeMissing[{}]", exclude_index),
362                        field.to_string(),
363                    ));
364                    exclude_index += 1;
365                }
366            }
367        }
368
369        pairs
370    }

373/// Combine the plain field lists from the `-v` flag and the field-builder flags.
374///
375/// Both inputs are plain comma-separated lists (no prefix, no encoding).
376fn combine_variable_string(v: String, fb: String) -> String {
377    match (v.is_empty(), fb.is_empty()) {
378        (true, true) => "".into(),
379        (true, false) => fb,
380        (false, true) => v,
381        (false, false) => format!("{},{}", v, fb),
382    }
383}

385/// Build the `query` value from its parts.
386///
387/// An empty `taxon` makes a query across all taxa. `tax_rank` and
388/// `expression` are as returned by [`crate::utils::tax_ranks::TaxRanks::parse`]
389/// and [`format_expression`], i.e. starting with `" AND "`. As `OR` binds
390/// more loosely than `AND` in the API, the taxon and rank are repeated in
391/// each `OR` branch of the expression, so they apply to all of them.
392pub fn build_query(taxon: &str, tax_tree: &str, tax_rank: &str, expression: &str) -> String {
393    let taxon_term = if taxon.is_empty() {
394        String::new()
395    } else {
396        format!("tax_{}({})", tax_tree, taxon)
397    };
398    let tax_rank = tax_rank.trim_start_matches(" AND ");
399    let expression = expression.trim_start_matches(" AND ");
400
401    expression
402        .split(" OR ")
403        .map(|group| {
404            [taxon_term.as_str(), tax_rank, group]
405                .into_iter()
406                .filter(|part| !part.is_empty())
407                .collect::<Vec<_>>()
408                .join(" AND ")
409        })
410        .collect::<Vec<_>>()
411        .join(" OR ")
412}

414/// The `sortBy`/`sortOrder` URL parameters (with a leading `&`) for
415/// `--sort-by`/`--sort-order`.
416///
417/// GoaT ignores a field it doesn't know, so the field is checked here: it
418/// must be a field of the index (synonyms are resolved) or one of the
419/// record's own fields.
420pub fn sort_params(sort_by: &str, order: &str, index_type: IndexType) -> Result<String> {
421    let builtin: &[&str] = match index_type {
422        IndexType::Taxon => &["scientific_name", "taxon_id", "taxon_rank"],
423        IndexType::Assembly => &["scientific_name", "taxon_id", "assembly_id"],
424    };
425    let field = sort_by.trim();
426    let field = if builtin.contains(&field) {
427        field.to_string()
428    } else {
429        let data = match index_type {
430            IndexType::Taxon => &*GOAT_TAXON_VARIABLE_DATA,
431            IndexType::Assembly => &*GOAT_ASSEMBLY_VARIABLE_DATA,
432        };
433        Variables::new(field).parse_one(data)?
434    };
435    Ok(format!(
436        "&sortBy={}&sortOrder={}",
437        percent_encode_query_value(&field),
438        order
439    ))
440}

442/// The function which creats the GoaT API URLs
443/// which are then used as GET requests.
444pub fn make_goat_urls(
445    api: &str,
446    taxids: &[String],
447    goat_url: &str,
448    tax_tree: &str,
449    include_estimates: bool,
450    include_raw_values: bool,
451    exclude: bool,
452    summarise_values_by: &str,
453    result: &str,
454    taxonomy: &str,
455    size: u64,
456    ranks: &str,
457    fields: FieldBuilder,
458    variables: Option<&str>,
459    expression: &str,
460    tax_rank: &str,
461    unique_ids: Vec<String>,
462    index_type: IndexType,
463) -> Result<Vec<String>> {
464    // make the rank string
465    let rank_string = format_rank(ranks);
466
467    // parse the variables, if they have been given.
468    let variables_field_string = if let Some(variables) = variables {
469        match index_type {
470            IndexType::Taxon => Variables::new(variables)
471                .parse(&GOAT_TAXON_VARIABLE_DATA, fields.taxon_toggle_direct)?,
472            IndexType::Assembly => Variables::new(variables)
473                .parse(&GOAT_ASSEMBLY_VARIABLE_DATA, fields.taxon_toggle_direct)?,
474        }
475    } else {
476        "".into()
477    };
478
479    // parse the fields from the flags
480    let field_builder_string = fields.build_fields_string();
481    // and combine
482    let fields_string = combine_variable_string(variables_field_string, field_builder_string);
483
484    let names_string = format_names(fields.taxon_names);
485
486    let exclude_pairs: Vec<(String, String)> = if exclude {
487        match variables {
488            Some(v) => match index_type {
489                IndexType::Taxon => Variables::new(v).parse_exclude(&GOAT_TAXON_VARIABLE_DATA)?,
490                IndexType::Assembly => {
491                    Variables::new(v).parse_exclude(&GOAT_ASSEMBLY_VARIABLE_DATA)?
492                }
493            },
494            None => fields.generate_exclude_flags(),
495        }
496    } else {
497        vec![]
498    };
499
500    // enumeration of the taxa will be 0 -> n,
501    // corresponding to alphabetical order of taxa
502    let mut res = Vec::new();
503    for (taxon, query_id_suffix) in taxids.iter().zip(unique_ids.iter()) {
504        // Build the GoaT query language value (plain text; encoded below)
505        let query_value = build_query(taxon, tax_tree, tax_rank, expression);
506
507        // Build the raw query string manually so that spaces are encoded as
508        // %20 (not +).  url::query_pairs_mut uses form-encoding which GoaT rejects.
509        let mut qp: Vec<String> = Vec::new();
510        qp.push(format!("query={}", percent_encode_query_value(&query_value)));
511        qp.push(format!("includeEstimates={}", include_estimates));
512        qp.push(format!("includeRawValues={}", include_raw_values));
513        qp.push(format!("summaryValues={}", summarise_values_by));
514        qp.push(format!("result={}", result));
515        qp.push(format!("taxonomy={}", taxonomy));
516        qp.push(format!("size={}", size));
517        if !rank_string.is_empty() {
518            qp.push(format!("ranks={}", percent_encode_query_value(&rank_string)));
519        }
520        if !fields_string.is_empty() {
521            qp.push(format!("fields={}", percent_encode_query_value(&fields_string)));
522        }
523        if fields.taxon_tidy {
524            qp.push("tidyData=true".into());
525        }
526        if !names_string.is_empty() {
527            qp.push(format!("names={}", percent_encode_query_value(&names_string)));
528        }
529        qp.push(format!("queryId=goat_cli_{}", query_id_suffix));
530        for (key, value) in &exclude_pairs {
531            qp.push(format!("{}={}", key, percent_encode_query_value(value)));
532        }
533
534        let base = format!("{}{}", goat_url, api);
535        let mut url = Url::parse(&base).expect("goat_url is a valid base");
536        url.set_query(Some(&qp.join("&")));
537        res.push(url.to_string());
538    }
539    Ok(res)
540}
