12/// Summary functions that can wrap a field in an expression, e.g.
13/// `max(genome_size)`. Mirrors the API's own list (`functions/summaries.js`),
14/// minus its internal `metadata` and `hexbin*` summaries.
15pub const VALID_EXPRESSION_FUNCTIONS: &[&str] =
16    &["min", "max", "count", "length", "sp_count", "range", "value"];
17
18/// Functions which return a count, whatever the type of the field.
19const COUNT_FUNCTIONS: &[&str] = &["count", "length", "sp_count"];
20
21/// Aggregation subset specifiers that may follow a field name with a colon.
22const VALID_SUBSETS: &[&str] = &["direct", "ancestor", "descendant", "estimate"];

24/// The type checked against the value of a count function.
25static COUNT_TYPE: TypeOf<'static> = TypeOf::Long;

27/// The API splits on `or`, then `and`, case-insensitively and only where
28/// surrounded by whitespace (so values such as `island` are safe).
29static OR_SPLIT: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"(?i)\s+or\s+").unwrap());
30static AND_SPLIT: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"(?i)\s+and\s+").unwrap());
31/// `<lhs> <operator> <rhs>`. Two character operators must come first.
32static CLAUSE: LazyLock<Regex> = LazyLock::new(|| {
33    Regex::new(r"^(?P<lhs>[^!<>=]+?)\s*(?P<op>!=|<=|>=|==|<|>|=)\s*(?P<rhs>.*)$").unwrap()
34});
35/// A summary function wrapping a field, e.g. `max(genome_size:direct)`.
36static FUNCTION: LazyLock<Regex> =
37    LazyLock::new(|| Regex::new(r"^(?P<func>[a-z_]+)\(\s*(?P<field>[^()]*?)\s*\)$").unwrap());
38/// A number with a size suffix, e.g. `1G`, which the API does not accept.
39static NUMBER_WITH_SUFFIX: LazyLock<Regex> =
40    LazyLock::new(|| Regex::new(r"(?i)^-?\d+(\.\d+)?\s*[kmgtp]b?$").unwrap());

46/// Serialize GoaT variables into their types.
47///
48/// See [here](https://www.elastic.co/guide/en/elasticsearch/reference/current/number.html)
49/// for more details.
50#[derive(Tabled)]
51pub enum TypeOf<'a> {
52    /// Signed 64 bit int.
53    Long,
54    /// Signed 16 bit int.
55    Short,
56    /// Float with one decimal place.
57    OneDP,
58    /// Float with two decimal places.
59    TwoDP,
60    /// Signed 32 bit int.
61    Integer,
62    /// A date.
63    Date,
64    /// Half precision 16 bit float.
65    HalfFloat,
66    /// A keyword. If the list of values is not empty, the API rejects
67    /// any other value.
68    Keyword(Vec<&'a str>),
69    /// A keyword with a list of known values, which the API does not
70    /// enforce; other values are allowed but may match nothing.
71    KeywordSuggest(Vec<&'a str>),
72    /// Not usable in expressions (e.g. `geo_point`), so not checked.
73    None,
74}
75
76impl<'a> TypeOf<'a> {
77    /// Check the right hand side of a clause, so `goat-cli` displays
78    /// meaningful help before the query is sent.
79    ///
80    /// `rhs` may be a comma separated list, each value may be negated with a
81    /// leading `!`, and `null` is allowed for any type.
82    fn check(&self, rhs: &str, variable: &str) -> Result<()> {
83        for value in rhs.split(',') {
84            let value = value.trim().trim_start_matches('!').trim();
85            if value.is_empty() {
86                return Err(expression_error(format!(
87                    "missing value for \"{variable}\" in \"{rhs}\"."
88                )));
89            }
90            if value.eq_ignore_ascii_case("null") {
91                continue;
92            }
93            match self {
94                TypeOf::Long
95                | TypeOf::Short
96                | TypeOf::OneDP
97                | TypeOf::TwoDP
98                | TypeOf::Integer
99                | TypeOf::HalfFloat => check_number(value, variable)?,
100                TypeOf::Date => check_date(value, variable)?,
101                TypeOf::Keyword(allowed) if !allowed.is_empty() => {
102                    if let Some(suggestion) = unknown_keyword(value, allowed) {
103                        return Err(expression_error(format!(
104                            "\"{value}\" is not a valid value for \"{variable}\" - did you mean \"{suggestion}\"?"
105                        )));
106                    }
107                }
108                TypeOf::KeywordSuggest(known) => {
109                    if let Some(suggestion) = unknown_keyword(value, known) {
110                        eprintln!(
111                            "warning: \"{value}\" is not a known value for \"{variable}\" (did you mean \"{suggestion}\"?), so may match nothing."
112                        );
113                    }
114                }
115                TypeOf::Keyword(_) | TypeOf::None => (),
116            }
117        }
118        Ok(())
119    }
120}

122/// The API accepts anything JavaScript parses as a number, including
123/// scientific notation (`1e9`) and negative numbers.
124fn check_number(value: &str, variable: &str) -> Result<()> {
125    if value
126        .replace('−', "-")
127        .parse::<f64>()
128        .map_or(false, f64::is_finite)
129    {
130        return Ok(());
131    }
132    let hint = if NUMBER_WITH_SUFFIX.is_match(value) {
133        " Size suffixes are not supported; use scientific notation instead, e.g. 1e9."
134    } else {
135        ""
136    };
137    Err(expression_error(format!(
138        "for variable \"{variable}\", \"{value}\" is not a number.{hint}"
139    )))
140}

142/// Dates may be given as `yyyy`, `yyyy-mm` or `yyyy-mm-dd`.
143fn check_date(value: &str, variable: &str) -> Result<()> {
144    let digits = |s: &str, n: usize| s.len() == n && s.chars().all(|c| c.is_ascii_digit());
145    let in_range = |s: &str, max: u32| s.parse::<u32>().map_or(false, |v| (1..=max).contains(&v));
146    let valid = match value.split('-').collect::<Vec<_>>().as_slice() {
147        [y] => digits(y, 4),
148        [y, m] => digits(y, 4) && digits(m, 2) && in_range(m, 12),
149        [y, m, d] => {
150            digits(y, 4) && digits(m, 2) && in_range(m, 12) && digits(d, 2) && in_range(d, 31)
151        }
152        _ => false,
153    };
154    if valid {
155        Ok(())
156    } else {
157        Err(expression_error(format!(
158            "for variable \"{variable}\", \"{value}\" is not a date. Use yyyy, yyyy-mm or yyyy-mm-dd."
159        )))
160    }
161}

163/// If `value` is not (case-insensitively) one of `allowed`, return the
164/// closest match.
165fn unknown_keyword(value: &str, allowed: &[&str]) -> Option<String> {
166    if allowed.iter().any(|a| a.eq_ignore_ascii_case(value)) {
167        return None;
168    }
169    let possibilities = allowed.iter().map(|a| a.to_string()).collect::<Vec<_>>();
170    Some(did_you_mean(&possibilities, value).unwrap_or_default())
171}

173impl<'a> fmt::Display for TypeOf<'a> {
174    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
175        match self {
176            // do nothing with None at the moment.
177            TypeOf::None => write!(f, "Please don't use yet! This variable needs fixing."),
178            TypeOf::Long
179            | TypeOf::Short
180            | TypeOf::OneDP
181            | TypeOf::TwoDP
182            | TypeOf::Integer
183            | TypeOf::Date
184            | TypeOf::HalfFloat => write!(f, "!=, <, <=, =, ==, >, >="),
185            TypeOf::Keyword(k) | TypeOf::KeywordSuggest(k) if k.is_empty() => write!(f, ""),
186            TypeOf::Keyword(k) | TypeOf::KeywordSuggest(k) => write!(f, "== {}", k.join(", ")),
187        }
188    }
189}

191/// Resolve a field name as the API does: case-insensitively, accepting
192/// synonyms (e.g. `ebp_metric_date`) and `-` in place of `_`.
193pub fn canonical_field(
194    name: &str,
195    reference_data: &BTreeMap<&'static str, Variable<'static>>,
196) -> Option<&'static str> {
197    let lookup = |n: &str| {
198        reference_data
199            .keys()
200            .find(|k| k.eq_ignore_ascii_case(n))
201            .copied()
202    };
203    lookup(name)
204        .or_else(|| lookup(&name.replace('-', "_")))
205        .or_else(|| {
206            GOAT_VARIABLE_SYNONYMS
207                .get(name.to_lowercase().as_str())
208                .and_then(|canonical| lookup(canonical))
209        })
210}

244/// Print the table of GoaT variable data.
245pub fn print_variable_data(data: &BTreeMap<&'static str, Variable<'static>>) -> Result<()> {
246    // for some space
247    crate::outln!()?;
248    // map the header to a tuple combination
249    // see https://github.com/zhiburt/tabled/blob/master/README.md
250    let table_data = data
251        .iter()
252        .map(|(e, f)| (ColHeader(e), f))
253        .collect::<Vec<(ColHeader, &Variable)>>();
254    // add taxon ranks at end...
255    let footer_data = TaxRanks::init();
256
257    let table_string = Table::new(&table_data)
258        .with(Panel::footer(format!(
259            "NCBI taxon ranks:\n\n{}",
260            footer_data
261        )))
262        .with(Modify::new(Rows::new(1..table_data.len() - 1)).with(Width::wrap(30).keep_words()))
263        // 4 rows
264        .with(Modify::new(Rows::new(table_data.len()..)).with(Width::wrap(30 * 4).keep_words()))
265        .to_string();
266
267    crate::outln!("{}", table_string)?;
268    Ok(())
269}

271/// The CLI expression which needs to be parsed.
272///
273/// See `EXPRESSIONS.md` for the syntax.

275pub struct CLIexpression<'a> {
276    pub inner: &'a str,
277}
278
279impl<'a> CLIexpression<'a> {
280    /// Constructor for [`CLIexpression`].
281    pub fn new(string: &'a str) -> Self {
282        Self { inner: string }
283    }
284
285    /// Validate an expression and return it in canonical form, as
286    /// `" AND <clause> AND <clause> OR <clause> ..."`, ready to follow a
287    /// taxon term.
288    ///
289    /// As in the API, `OR` binds more loosely than `AND`, so the result is a
290    /// list of `AND` groups joined by `" OR "`. The taxon and rank must apply
291    /// to every group; [`crate::utils::url::make_goat_urls`] does that.
292    pub fn parse(
293        &mut self,
294        reference_data: &BTreeMap<&'static str, Variable<'static>>,
295        extra_fields: Option<&HashSet<String>>,
296    ) -> Result<String> {
297        let input = self.inner.trim();
298        if input.is_empty() {
299            return Err(expression_error("the expression is empty.".to_string()));
300        }
301        if input.contains("&&") {
302            return Err(expression_error(
303                "use the AND keyword, not &&, between clauses.".to_string(),
304            ));
305        }
306        if input.contains("||") {
307            return Err(expression_error(
308                "use the OR keyword, not ||, between clauses.".to_string(),
309            ));
310        }
311
312        let mut groups = Vec::new();
313        for group in OR_SPLIT.split(input) {
314            let clauses = AND_SPLIT
315                .split(strip_outer_parens(group.trim()))
316                .map(|clause| parse_clause(clause, reference_data, extra_fields))
317                .collect::<Result<Vec<_>>>()?;
318            groups.push(clauses.join(" AND "));
319        }
320        Ok(format!(" AND {}", groups.join(" OR ")))
321    }
322
323    /// Whether any clause names a field that `reference_data` does not know,
324    /// which is when the live field registry is worth loading. Unlike
325    /// [`CLIexpression::parse`], this prints no warnings.
326    pub fn has_unknown_field(
327        &self,
328        reference_data: &BTreeMap<&'static str, Variable<'static>>,
329    ) -> bool {
330        OR_SPLIT
331            .split(self.inner.trim())
332            .flat_map(|group| AND_SPLIT.split(strip_outer_parens(group.trim())))
333            .any(|clause| {
334                let clause = clause.replace(['"', '\''], "");
335                let clause = clause.trim();
336                let lhs = CLAUSE
337                    .captures(clause)
338                    .map_or(clause, |caps| caps.name("lhs").unwrap().as_str());
339                !lhs.trim().to_lowercase().starts_with("tax_")
340                    && resolve_lhs(lhs, reference_data, None).is_err()
341            })
342    }
343}
344
345/// Remove parentheses wrapping a whole `OR` branch, e.g. `(a AND b)`.
346fn strip_outer_parens(group: &str) -> &str {
347    let Some(inner) = group.strip_prefix('(').and_then(|g| g.strip_suffix(')')) else {
348        return group;
349    };
350    // make sure the first `(` closes at the very end, not e.g. `(a) AND (b)`
351    let mut depth = 0i32;
352    for c in inner.chars() {
353        match c {
354            '(' => depth += 1,
355            ')' => depth -= 1,
356            _ => (),
357        }
358        if depth < 0 {
359            return group;
360        }
361    }
362    inner.trim()
363}

365/// Validate a single clause, returning it in canonical form.
366fn parse_clause(
367    clause: &str,
368    reference_data: &BTreeMap<&'static str, Variable<'static>>,
369    extra_fields: Option<&HashSet<String>>,
370) -> Result<String> {
371    // quotes are not needed by the API, even for values containing spaces
372    let clause = clause.replace(['"', '\''], "");
373    let clause = clause.trim();
374    if clause.is_empty() {
375        return Err(expression_error(
376            "found an empty clause - check for a doubled or trailing AND/OR.".to_string(),
377        ));
378    }
379    if clause.matches('(').count() != clause.matches(')').count() {
380        return Err(expression_error(format!(
381            "unbalanced parentheses in \"{clause}\". Parentheses may only wrap a whole OR branch, as GoaT does not support nesting: write \"a AND (b OR c)\" as \"a AND b OR a AND c\"."
382        )));
383    }
384    let lower = clause.to_lowercase();
385    if lower.starts_with("tax_rank") {
386        return Err(expression_error(
387            "set tax_rank through --tax-rank <taxon_rank>.".to_string(),
388        ));
389    }
390    if lower.starts_with("tax_") {
391        return Err(expression_error("set tax_name through -t <taxon_name>, tax_tree by -d flag, and tax_lineage by -l flag.".to_string()));
392    }
393
394    let Some(caps) = CLAUSE.captures(clause) else {
395        // a bare field, which means "has a value"
396        if clause.contains(char::is_whitespace) {
397            return Err(expression_error(format!(
398                "\"{clause}\" is not a valid clause. Expected <field> <operator> <value>, e.g. \"long_list = dtol\", or a bare field name."
399            )));
400        }
401        let (lhs, _) = resolve_lhs(clause, reference_data, extra_fields)?;
402        return Ok(lhs);
403    };
404
405    let (lhs, type_of) = resolve_lhs(&caps["lhs"], reference_data, extra_fields)?;
406    let rhs = caps["rhs"].trim();
407    if rhs.is_empty() {
408        return Err(expression_error(format!("missing value in \"{clause}\".")));
409    }
410    if let Some(type_of) = type_of {
411        type_of.check(rhs, &lhs)?;
412    }
413    let values = rhs.split(',').map(str::trim).collect::<Vec<_>>().join(",");
414    Ok(format!("{} {} {}", lhs, &caps["op"], values))
415}

417/// Validate the left hand side of a clause: a field, optionally with a
418/// `:subset` and/or wrapped in a summary function. Returns the canonical
419/// form, and the type to check values against (`None` for fields only known
420/// from the live registry, and identifiers such as `taxon_id`).
421fn resolve_lhs<'d>(
422    lhs: &str,
423    reference_data: &'d BTreeMap<&'static str, Variable<'static>>,
424    extra_fields: Option<&HashSet<String>>,
425) -> Result<(String, Option<&'d TypeOf<'static>>)> {
426    let lhs = lhs.trim().to_lowercase();
427    let (function, field) = match FUNCTION.captures(&lhs) {
428        Some(caps) => (Some(caps["func"].to_string()), caps["field"].to_string()),
429        None => (None, lhs.clone()),
430    };
431    if let Some(f) = &function {
432        if !VALID_EXPRESSION_FUNCTIONS.contains(&f.as_str()) {
433            return Err(expression_error(format!(
434                "unknown function \"{f}\" in \"{lhs}\" - valid functions are: {}.",
435                VALID_EXPRESSION_FUNCTIONS.join(", ")
436            )));
437        }
438    }
439
440    let (name, subset) = match field.split_once(':') {
441        Some((name, subset)) => (name.trim(), Some(subset.trim())),
442        None => (field.as_str(), None),
443    };
444    if let Some(subset) = subset {
445        if !VALID_SUBSETS.contains(&subset) {
446            return Err(expression_error(format!(
447                "unknown field subset \":{}\" - valid subsets are: {}",
448                subset,
449                VALID_SUBSETS.join(", ")
450            )));
451        }
452    }
453    if name.is_empty() || name.contains(|c: char| c.is_whitespace() || c == '(' || c == ')') {
454        return Err(expression_error(format!(
455            "\"{lhs}\" is not a valid field name."
456        )));
457    }
458
459    let is_dynamic = extra_fields.map_or(false, |fields| fields.contains(name));
460    let (name, type_of) = match canonical_field(name, reference_data) {
461        Some(canonical) => (canonical.to_string(), Some(&reference_data[canonical].type_of)),
462        // identifiers are matched by the API, e.g. taxon_id = 9606
463        None if is_dynamic || name.ends_with("_id") => (name.to_string(), None),
464        None => {
465            let possibilities = reference_data
466                .keys()
467                .map(|k| k.to_string())
468                .chain(GOAT_VARIABLE_SYNONYMS.keys().map(|k| k.to_string()))
469                .collect::<Vec<_>>();
470            let hint = did_you_mean(&possibilities, name)
471                .map(|s| format!(" - did you mean \"{s}\"?"))
472                .unwrap_or_default();
473            return Err(expression_error(format!(
474                "unknown variable \"{name}\"{hint}"
475            )));
476        }
477    };
478
479    let type_of = match function.as_deref() {
480        Some(f) if COUNT_FUNCTIONS.contains(&f) => Some(&COUNT_TYPE),
481        _ => type_of,
482    };
483    let field = match subset {
484        Some(subset) => format!("{name}:{subset}"),
485        None => name,
486    };
487    let lhs = match function {
488        Some(f) => format!("{f}({field})"),
489        None => field,
490    };
491    Ok((lhs, type_of))
492}
